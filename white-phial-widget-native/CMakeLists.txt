cmake_minimum_required(VERSION 3.24)
project(WhitePhialWidget VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT OR NOT DEFINED MENU_API_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT and MENU_API_ROOT to the pinned checkouts")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(WhitePhialWidget SHARED src/main.cpp src/Widget.h src/Config.h)
target_link_libraries(WhitePhialWidget PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(WhitePhialWidget PRIVATE src "${MENU_API_ROOT}")
target_compile_definitions(WhitePhialWidget PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(widget_tests tests/widget_tests.cpp)
target_include_directories(widget_tests PRIVATE src)
target_compile_definitions(widget_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(widget_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(widget_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME widget_state_settings_and_save_isolation COMMAND widget_tests)
