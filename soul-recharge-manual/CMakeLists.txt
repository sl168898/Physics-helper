cmake_minimum_required(VERSION 3.24)
project(SoulRechargeMenu VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
find_package(nlohmann_json CONFIG REQUIRED)
target_compile_options(CommonLibSSE PRIVATE /MP2)
add_library(SoulRechargeMenu SHARED src/main.cpp src/CoreBridge.h src/InputRules.h)
target_include_directories(SoulRechargeMenu PRIVATE vendor)
target_link_libraries(SoulRechargeMenu PRIVATE CommonLibSSE::CommonLibSSE nlohmann_json::nlohmann_json bcrypt)
target_compile_definitions(SoulRechargeMenu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
target_compile_definitions(core_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(core_tests PRIVATE /UNDEBUG)
target_link_libraries(core_tests PRIVATE bcrypt)
add_test(NAME original_kernel_and_hotkeys COMMAND core_tests "${CMAKE_CURRENT_SOURCE_DIR}/original/SKSE/Plugins/AGH_SoulRecharge_G0.dll")
