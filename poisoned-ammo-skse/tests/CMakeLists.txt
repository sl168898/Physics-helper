cmake_minimum_required(VERSION 3.24)
project(PoisonedAmmoCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(core_tests core_tests.cpp)
target_include_directories(core_tests PRIVATE ../src)
if(MSVC)
    target_compile_options(core_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(core_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME persistence_esl_and_batch_properties COMMAND core_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/make_impact_test.py"
    "${CMAKE_CURRENT_BINARY_DIR}/impact_abi_test.cpp" COMMAND_ERROR_IS_FATAL ANY)
add_executable(impact_abi_test "${CMAKE_CURRENT_BINARY_DIR}/impact_abi_test.cpp")
if(MSVC)
    target_compile_options(impact_abi_test PRIVATE /UNDEBUG)
else()
    target_compile_options(impact_abi_test PRIVATE -UNDEBUG)
endif()
add_test(NAME production_impact_pointer_survives_cleanup COMMAND impact_abi_test)
