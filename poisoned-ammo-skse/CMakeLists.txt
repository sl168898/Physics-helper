cmake_minimum_required(VERSION 3.24)
project(PoisonedAmmoNative VERSION 0.2.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT to the pinned CommonLibSSE-NG checkout")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(PoisonedAmmoNative SHARED src/main.cpp src/Core.h src/ImmersiveAnimation.h src/ReloadGate.h src/PoisonSnapshot.h src/PoisonKeywords.h)
target_link_libraries(PoisonedAmmoNative PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(PoisonedAmmoNative PRIVATE src)
target_compile_definitions(PoisonedAmmoNative PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(PoisonedAmmoNative PRIVATE /EHsc /W4)
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
target_compile_options(core_tests PRIVATE /UNDEBUG)
enable_testing()
add_test(NAME persistence_esl_and_batch_properties COMMAND core_tests)
