cmake_minimum_required(VERSION 3.22)
project(WheelerPotionBatchTests LANGUAGES CXX)
enable_testing()
add_executable(batch_tests batch_tests.cpp)
target_compile_features(batch_tests PRIVATE cxx_std_20)
target_include_directories(batch_tests PRIVATE "${REFINED_SOURCE}/src")
if(MSVC)
  target_compile_options(batch_tests PRIVATE /utf-8)
endif()
add_test(NAME batch_identity COMMAND batch_tests)

add_executable(icon_name_tests icon_name_tests.cpp)
target_compile_features(icon_name_tests PRIVATE cxx_std_20)
target_include_directories(icon_name_tests PRIVATE "${REFINED_SOURCE}/src")
if(MSVC)
  target_compile_options(icon_name_tests PRIVATE /utf-8)
endif()
add_test(NAME icon_name_identity COMMAND icon_name_tests)
