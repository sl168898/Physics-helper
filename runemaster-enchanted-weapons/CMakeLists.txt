cmake_minimum_required(VERSION 3.24)
project(RunemasterEnchantmentBridge VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
if(MSVC)
 target_compile_options(CommonLibSSE PRIVATE /MP2)
endif()
add_library(RunemasterEnchantmentBridge SHARED src/main.cpp src/Rules.h src/Crafting.cpp src/Crafting.h src/CraftingRules.h src/Visuals.cpp src/Visuals.h src/VisualRules.h)
target_link_libraries(RunemasterEnchantmentBridge PRIVATE CommonLibSSE::CommonLibSSE)
add_executable(rules_tests tests/rules_tests.cpp)
target_include_directories(rules_tests PRIVATE src)
if(MSVC)
 target_compile_options(rules_tests PRIVATE /UNDEBUG)
endif()
enable_testing()
add_test(NAME rune_rules COMMAND rules_tests)
add_executable(crafting_tests tests/crafting_tests.cpp)
target_include_directories(crafting_tests PRIVATE src)
if(MSVC)
 target_compile_options(crafting_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME crafting_preservation COMMAND crafting_tests)
add_executable(visual_tests tests/visual_tests.cpp)
target_include_directories(visual_tests PRIVATE src)
if(MSVC)
 target_compile_options(visual_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME weapon_visual_lifecycle COMMAND visual_tests)

if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
 add_executable(abi_tests tests/abi_tests.cpp)
 target_compile_options(abi_tests PRIVATE /UNDEBUG /Od)
 add_test(NAME inventory_hook_abi COMMAND abi_tests)
endif()
