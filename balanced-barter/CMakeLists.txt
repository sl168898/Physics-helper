cmake_minimum_required(VERSION 3.24)
project(BalancedBarter VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
if(MSVC)
 target_compile_options(CommonLibSSE PRIVATE /MP2)
endif()
add_library(BalancedBarter SHARED src/main.cpp src/TradeRules.h)
target_link_libraries(BalancedBarter PRIVATE CommonLibSSE::CommonLibSSE)
enable_testing()
add_executable(trade_tests tests/trade_tests.cpp)
target_include_directories(trade_tests PRIVATE src)
if(MSVC)
 target_compile_options(trade_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME settlement_and_rollback COMMAND trade_tests)
