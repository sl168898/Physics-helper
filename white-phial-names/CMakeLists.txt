cmake_minimum_required(VERSION 3.24)
project(WhitePhialNames VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(WhitePhialNames SHARED src/main.cpp src/Names.h)
target_link_libraries(WhitePhialNames PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(WhitePhialNames PRIVATE NOMINMAX)
add_executable(names_tests tests/names_tests.cpp)
target_include_directories(names_tests PRIVATE src)
if(MSVC)
    target_compile_options(names_tests PRIVATE /UNDEBUG /utf-8)
endif()
enable_testing()
add_test(NAME phial_name_preservation COMMAND names_tests)
