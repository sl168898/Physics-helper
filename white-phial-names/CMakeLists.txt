cmake_minimum_required(VERSION 3.24)
project(WhitePhialNames VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(WhitePhialNames SHARED src/main.cpp src/Blacklist.cpp src/Blacklist.h src/BlacklistCore.h src/Names.h src/Storage.cpp src/Storage.h src/Bank.h src/NamedKeywords.h src/CosaveFile.h)
target_link_libraries(WhitePhialNames PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(WhitePhialNames PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(WhitePhialNames PRIVATE "${MENU_API_ROOT}")
add_executable(names_tests tests/names_tests.cpp)
target_include_directories(names_tests PRIVATE src)
if(MSVC)
    target_compile_options(names_tests PRIVATE /UNDEBUG /utf-8)
endif()
enable_testing()
add_test(NAME phial_name_preservation COMMAND names_tests)

add_executable(bank_tests tests/bank_tests.cpp)
target_include_directories(bank_tests PRIVATE src)
if(MSVC)
    target_compile_options(bank_tests PRIVATE /UNDEBUG /utf-8)
endif()
add_test(NAME protected_liquid_bank COMMAND bank_tests)

add_executable(cosave_tests tests/cosave_tests.cpp)
target_include_directories(cosave_tests PRIVATE src)
if(MSVC)
    target_compile_options(cosave_tests PRIVATE /UNDEBUG /utf-8)
endif()
add_test(NAME protected_cosave_lookup COMMAND cosave_tests)

add_executable(blacklist_tests tests/blacklist_tests.cpp)
target_include_directories(blacklist_tests PRIVATE src)
target_compile_definitions(blacklist_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(blacklist_tests PRIVATE /UNDEBUG /utf-8)
endif()
add_test(NAME blacklist_identity_and_persistence COMMAND blacklist_tests)
