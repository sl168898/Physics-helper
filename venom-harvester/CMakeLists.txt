cmake_minimum_required(VERSION 3.24)
project(VenomHarvester VERSION 2.0.10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT to the pinned CommonLibSSE-NG checkout")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(VenomHarvester SHARED src/main.cpp src/Harvest.h src/HarvestYield.h src/MenuGate.h src/CraftCapture.h src/DeferredForms.h src/PendingCrafts.h src/InventoryBottleRefs.h src/DamageObservation.h)
target_link_libraries(VenomHarvester PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(VenomHarvester PRIVATE src)
add_executable(harvest_tests tests/harvest_tests.cpp)
target_include_directories(harvest_tests PRIVATE src)
if(MSVC)
    target_compile_options(harvest_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(harvest_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME poison_harvest COMMAND harvest_tests)
add_executable(menu_tests tests/menu_tests.cpp)
target_include_directories(menu_tests PRIVATE src)
if(MSVC)
    target_compile_options(menu_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(menu_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_menu_lifecycle COMMAND menu_tests)
add_executable(crafting_tests tests/crafting_tests.cpp)
target_include_directories(crafting_tests PRIVATE src)
if(MSVC)
    target_compile_options(crafting_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(crafting_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_crafting_capture COMMAND crafting_tests)
add_executable(inventory_event_tests tests/inventory_event_tests.cpp)
target_include_directories(inventory_event_tests PRIVATE src)
if(MSVC)
    target_compile_options(inventory_event_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(inventory_event_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_inventory_event_lifetime COMMAND inventory_event_tests)
add_executable(pending_crafts_tests tests/pending_crafts_tests.cpp)
target_include_directories(pending_crafts_tests PRIVATE src)
if(MSVC)
    target_compile_options(pending_crafts_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(pending_crafts_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_pending_crafts COMMAND pending_crafts_tests)
add_executable(inventory_ownership_tests tests/inventory_ownership_tests.cpp)
target_include_directories(inventory_ownership_tests PRIVATE src)
if(MSVC)
    target_compile_options(inventory_ownership_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(inventory_ownership_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_inventory_bottle_ownership COMMAND inventory_ownership_tests)
add_executable(damage_observation_tests tests/damage_observation_tests.cpp)
target_include_directories(damage_observation_tests PRIVATE src)
if(MSVC)
    target_compile_options(damage_observation_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(damage_observation_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_native_damage_observation COMMAND damage_observation_tests)
add_executable(harvest_yield_tests tests/harvest_yield_tests.cpp)
target_include_directories(harvest_yield_tests PRIVATE src)
if(MSVC)
    target_compile_options(harvest_yield_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(harvest_yield_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME satchel_double_harvest_yield COMMAND harvest_yield_tests)
install(TARGETS VenomHarvester RUNTIME DESTINATION SKSE/Plugins)
