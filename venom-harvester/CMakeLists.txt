cmake_minimum_required(VERSION 3.24)
project(VenomHarvester VERSION 2.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT to the pinned CommonLibSSE-NG checkout")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
if(MSVC)
    target_compile_options(CommonLibSSE PRIVATE /MP2)
endif()
add_library(VenomHarvester SHARED src/main.cpp src/Harvest.h src/MenuGate.h src/CraftCapture.h src/DeferredForms.h src/PendingCrafts.h src/InventoryBottleRefs.h src/DamageObservation.h)
target_link_libraries(VenomHarvester PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(VenomHarvester PRIVATE src)
add_executable(harvest_tests tests/harvest_tests.cpp)
target_include_directories(harvest_tests PRIVATE src)
if(MSVC)
    target_compile_options(harvest_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(harvest_tests PRIVATE -UNDEBUG)
endif()
