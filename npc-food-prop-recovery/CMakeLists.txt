cmake_minimum_required(VERSION 3.24)
project(NPCFoodPropRecovery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
if(MSVC)
  target_compile_options(CommonLibSSE PRIVATE /MP2)
endif()
add_library(NPCFoodPropRecovery SHARED src/main.cpp src/RecoveryPolicy.h)
target_link_libraries(NPCFoodPropRecovery PRIVATE CommonLibSSE::CommonLibSSE)
enable_testing()
add_executable(recovery_tests tests/recovery_tests.cpp)
target_include_directories(recovery_tests PRIVATE src)
if(MSVC)
  target_compile_options(recovery_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(recovery_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME recovery_lifecycle COMMAND recovery_tests)
