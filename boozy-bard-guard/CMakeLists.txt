cmake_minimum_required(VERSION 3.24)
project(BoozyBardGuard VERSION 1.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT)
 message(FATAL_ERROR "Set COMMONLIB_ROOT")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(BoozyBardGuard SHARED src/main.cpp src/Policy.h)
target_link_libraries(BoozyBardGuard PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(BoozyBardGuard PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(guard_tests tests/policy_tests.cpp)
target_include_directories(guard_tests PRIVATE src)
if(MSVC)
 target_compile_options(guard_tests PRIVATE /UNDEBUG)
else()
 target_compile_options(guard_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME alcohol_stacking_and_expiry COMMAND guard_tests)
