cmake_minimum_required(VERSION 3.24)
project(BatchCrafting VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT OR NOT DEFINED MENU_API_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT and MENU_API_ROOT to the pinned checkouts")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(BatchCrafting SHARED src/main.cpp src/Batch.h)
target_link_libraries(BatchCrafting PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(BatchCrafting PRIVATE src "${MENU_API_ROOT}")
target_compile_definitions(BatchCrafting PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(batch_tests tests/batch_tests.cpp)
target_include_directories(batch_tests PRIVATE src)
if(MSVC)
    target_compile_options(batch_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(batch_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME batch_safety_and_material_limits COMMAND batch_tests)
