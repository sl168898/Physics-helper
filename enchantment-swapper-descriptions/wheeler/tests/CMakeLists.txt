cmake_minimum_required(VERSION 3.22)
project(WheelerPotionBatchTests LANGUAGES CXX)
enable_testing()
add_executable(batch_tests batch_tests.cpp)
target_compile_features(batch_tests PRIVATE cxx_std_20)
target_include_directories(batch_tests PRIVATE "${REFINED_SOURCE}/src")
if(MSVC)
  target_compile_options(batch_tests PRIVATE /utf-8)
endif()
add_test(NAME batch_identity COMMAND batch_tests)
