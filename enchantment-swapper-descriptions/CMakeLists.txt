cmake_minimum_required(VERSION 3.24)
project(EnchantmentSwapperDescriptions VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
find_package(minhook CONFIG REQUIRED)
add_library(EnchantmentSwapperDescriptions SHARED src/main.cpp src/Bindings.h)
target_link_libraries(EnchantmentSwapperDescriptions PRIVATE CommonLibSSE::CommonLibSSE minhook::minhook)
target_compile_definitions(EnchantmentSwapperDescriptions PRIVATE NOMINMAX)
add_executable(bindings_tests tests/bindings_tests.cpp)
target_include_directories(bindings_tests PRIVATE src)
if(MSVC)
    target_compile_options(bindings_tests PRIVATE /UNDEBUG)
endif()
enable_testing()
add_test(NAME description_bindings COMMAND bindings_tests)
