cmake_minimum_required(VERSION 3.24)
project(BiggieTraitMechanics VERSION 1.4.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
find_package(minhook CONFIG REQUIRED)
add_library(BiggieTraitMechanics SHARED src/main.cpp src/Rules.h src/Skald.h src/SkaldRuntime.h src/SkaldDiagnostics.h src/EchoDiagnostics.h src/IronLungs.h src/IronLungsRuntime.h src/IronLungsGrant.h src/Overexertion.h)
target_link_libraries(BiggieTraitMechanics PRIVATE CommonLibSSE::CommonLibSSE minhook::minhook)
target_compile_definitions(BiggieTraitMechanics PRIVATE NOMINMAX)
add_executable(rules_tests tests/rules_tests.cpp)
target_include_directories(rules_tests PRIVATE src)
if(MSVC)
 target_compile_options(rules_tests PRIVATE /UNDEBUG)
endif()
enable_testing()
add_test(NAME traits_rules COMMAND rules_tests)
add_executable(lab_tests tests/lab_tests.cpp)
target_include_directories(lab_tests PRIVATE src)
if(MSVC)
 target_compile_options(lab_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME lab_visit_rules COMMAND lab_tests)
add_executable(skald_tests tests/skald_tests.cpp)
target_include_directories(skald_tests PRIVATE src)
if(MSVC)
 target_compile_options(skald_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME skald_recovery_and_echo COMMAND skald_tests)
add_executable(iron_lungs_tests tests/iron_lungs_tests.cpp)
target_include_directories(iron_lungs_tests PRIVATE src)
if(MSVC)
 target_compile_options(iron_lungs_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME iron_lungs_cost_threshold_and_snapshot COMMAND iron_lungs_tests)
add_executable(overexertion_tests tests/overexertion_tests.cpp)
target_include_directories(overexertion_tests PRIVATE src)
if(MSVC)
 target_compile_options(overexertion_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME iron_lungs_overexertion COMMAND overexertion_tests)
