cmake_minimum_required(VERSION 3.24)
project(StandingOvation VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT to the pinned CommonLibSSE-NG checkout")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(StandingOvation SHARED src/main.cpp src/RewardState.h)
target_link_libraries(StandingOvation PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(StandingOvation PRIVATE src)
add_executable(reward_tests tests/reward_tests.cpp)
target_include_directories(reward_tests PRIVATE src)
enable_testing()
add_test(NAME standing_ovation_reward_lifecycle COMMAND reward_tests)
