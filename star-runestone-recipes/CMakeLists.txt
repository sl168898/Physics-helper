cmake_minimum_required(VERSION 3.24)
project(StarRunestoneRecipes VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
if(MSVC)
 target_compile_options(CommonLibSSE PRIVATE /MP2)
endif()
add_library(StarRunestoneRecipes SHARED src/main.cpp src/CraftingRules.h)
target_link_libraries(StarRunestoneRecipes PRIVATE CommonLibSSE::CommonLibSSE)
enable_testing()
add_executable(crafting_tests tests/crafting_tests.cpp)
target_include_directories(crafting_tests PRIVATE src)
if(MSVC)
 target_compile_options(crafting_tests PRIVATE /UNDEBUG)
endif()
add_test(NAME soul_crafting_transactions COMMAND crafting_tests)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
 add_executable(abi_tests tests/abi_tests.cpp)
 target_compile_options(abi_tests PRIVATE /UNDEBUG /Od)
 add_test(NAME inventory_hook_abi COMMAND abi_tests)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME binary_plugin_records COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_plugin.py")
