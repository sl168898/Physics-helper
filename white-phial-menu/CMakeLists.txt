cmake_minimum_required(VERSION 3.24)
project(WhitePhialMenu VERSION 1.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
if(NOT DEFINED COMMONLIB_ROOT OR NOT DEFINED MENU_API_ROOT)
    message(FATAL_ERROR "Set COMMONLIB_ROOT and MENU_API_ROOT to the pinned checkouts")
endif()
add_subdirectory("${COMMONLIB_ROOT}" CommonLibSSE)
add_library(WhitePhialMenu SHARED src/main.cpp src/Settings.h src/Keys.h src/Profile.h src/ProfileStore.h)
target_link_libraries(WhitePhialMenu PRIVATE CommonLibSSE::CommonLibSSE)
target_include_directories(WhitePhialMenu PRIVATE src "${MENU_API_ROOT}")
target_compile_definitions(WhitePhialMenu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(settings_tests tests/settings_tests.cpp)
target_include_directories(settings_tests PRIVATE src)
add_executable(profile_tests tests/profile_tests.cpp)
target_include_directories(profile_tests PRIVATE src)
target_compile_definitions(profile_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(settings_tests PRIVATE /UNDEBUG)
    target_compile_options(profile_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(settings_tests PRIVATE -UNDEBUG)
    target_compile_options(profile_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME settings_and_save_isolation COMMAND settings_tests)
add_test(NAME shared_profile_persistence COMMAND profile_tests)
